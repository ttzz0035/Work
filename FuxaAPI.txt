REM ============================================================
REM 現在値取得：1タグ
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/getTagValue" -H "x-access-token: YOUR_TOKEN" --data-urlencode "ids=[\"TAG_ID_1\"]"

REM ============================================================
REM 現在値取得：複数タグ
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/getTagValue" -H "x-access-token: YOUR_TOKEN" --data-urlencode "ids=[\"TAG_ID_1\",\"TAG_ID_2\"]"

REM ============================================================
REM 現在値書込み：数値
REM ============================================================
curl.exe -sS -X POST "http://127.0.0.1:1881/api/setTagValue" -H "x-access-token: YOUR_TOKEN" -H "Content-Type: application/json" -d "{\"tags\":[{\"id\":\"TAG_ID_1\",\"value\":123.45}]}"

REM ============================================================
REM 現在値書込み：複数タグ
REM ============================================================
curl.exe -sS -X POST "http://127.0.0.1:1881/api/setTagValue" -H "x-access-token: YOUR_TOKEN" -H "Content-Type: application/json" -d "{\"tags\":[{\"id\":\"TAG_ID_1\",\"value\":123.45},{\"id\":\"TAG_ID_2\",\"value\":true},{\"id\":\"TAG_ID_3\",\"value\":\"RUN\"}]}"

REM ============================================================
REM DAQ履歴取得
REM from/toはUnix時刻のミリ秒
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/daq" -H "x-access-token: YOUR_TOKEN" --data-urlencode "query={\"from\":1785685200000,\"to\":1785688800000,\"sids\":[\"TAG_ID_1\"]}"

REM ============================================================
REM DAQ履歴取得：複数タグ
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/daq" -H "x-access-token: YOUR_TOKEN" --data-urlencode "query={\"from\":1785685200000,\"to\":1785688800000,\"sids\":[\"TAG_ID_1\",\"TAG_ID_2\"]}"

REM ============================================================
REM DAQ履歴取得：全パラメーター指定
REM gid   : グラフ・ゲージ側の識別子
REM event : B=後方、F=前方
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/daq" -H "x-access-token: YOUR_TOKEN" --data-urlencode "query={\"gid\":\"GAUGE_ID\",\"from\":1785685200000,\"to\":1785688800000,\"event\":\"B\",\"sids\":[\"TAG_ID_1\",\"TAG_ID_2\"]}"

REM ============================================================
REM APIキーで現在値取得
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/getTagValue" -H "x-api-key: YOUR_API_KEY" --data-urlencode "ids=[\"TAG_ID_1\"]"

REM ============================================================
REM APIキーでタグ値書込み
REM ============================================================
curl.exe -sS -X POST "http://127.0.0.1:1881/api/setTagValue" -H "x-api-key: YOUR_API_KEY" -H "Content-Type: application/json" -d "{\"tags\":[{\"id\":\"TAG_ID_1\",\"value\":123.45}]}"

REM ============================================================
REM APIキーでDAQ履歴取得
REM ============================================================
curl.exe -sS -G "http://127.0.0.1:1881/api/daq" -H "x-api-key: YOUR_API_KEY" --data-urlencode "query={\"from\":1785685200000,\"to\":1785688800000,\"sids\":[\"TAG_ID_1\"]}"
